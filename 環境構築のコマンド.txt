-ROS2 humbleのインストール-

$ sudo apt update && sudo apt upgrade -y && sudo apt autoremove -y
$ cd ~ && locale && sudo apt update && sudo apt install locales && sudo locale-gen en_US en_US.UTF-8 && sudo update-locale LC_ALL=en_US.UTF-8 LANG=en_US.UTF-8 && export LANG=en_US.UTF-8 && locale && sudo apt install software-properties-common -y && sudo add-apt-repository universe -y && sudo apt update && sudo apt install curl -y && sudo curl -sSL https://raw.githubusercontent.com/ros/rosdistro/master/ros.key -o /usr/share/keyrings/ros-archive-keyring.gpg && echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/ros-archive-keyring.gpg] http://packages.ros.org/ros2/ubuntu $(. /etc/os-release && echo $UBUNTU_CODENAME) main" | sudo tee /etc/apt/sources.list.d/ros2.list > /dev/null && sudo apt update && sudo apt upgrade -y && sudo apt autoremove -y && sudo apt install ros-humble-desktop -y && sudo apt install ros-dev-tools -y && source /opt/ros/humble/setup.bash && echo "export ROS_DOMAIN_ID=25" >> ~/.bashrc && echo "source /opt/ros/humble/setup.bash" >> ~/.bashrc

$ cd ~/rover_ws && colcon build --symlink-install --cmake-clean-cache
$ echo "source ~/rover_ws/install/local_setup.bash" >> ~/.bashrc && cd

$ ls /dev/ttyUSB* # 接続されているUSBの一覧
$ sudo chmod 666 /dev/ttyUSB1 # USB1へ権限を与える
$ ros2 run micro_ros_agent micro_ros_agent serial --baudrate 115200 --dev /dev/ttyUSB1 # 起動コマンド
$ ros2 launch ms_launch micro-ros_megarover_launch.py # このコマンドを打てば一発でmicro-ROSを起動できる

-Realsense SDKの構築-
T265はv2.53.1までしか対応していないため，git cloneのところでバージョン指定，Jetsonには未構築
参考サイト：[https://github.com/IntelRealSense/realsense-ros/?tab=readme-ov-file#installation]
$ cd ~ && sudo apt-get update && sudo apt-get upgrade -y && sudo apt-get dist-upgrade -y && sudo apt-get install git -y && git clone https://github.com/IntelRealSense/librealsense.git -b v2.53.1 && cd librealsense && sudo apt-get install libssl-dev libusb-1.0-0-dev libudev-dev pkg-config libgtk-3-dev cmake -y && sudo apt-get install libglfw3-dev libgl1-mesa-dev libglu1-mesa-dev at -y && sudo ./scripts/setup_udev_rules.sh -y && sudo ./scripts/patch-realsense-ubuntu-lts-hwe.sh && echo 'hid_sensor_custom' | sudo tee -a /etc/modules && mkdir build && cd build && cmake ../ -DCMAKE_BUILD_TYPE=Release -DBUILD_EXAMPLES=true -DFORCE_RSUSB_BACKEND=true -DBUILD_PYTHON_BINDINGS:bool=true -DBUILD_GRAPHICAL_EXAMPLES=true && sudo make uninstall && make clean && make -j12 && sudo make install
